```python
# File: src/predict.py
"""Prediction logic for wine price model."""

import joblib
import numpy as np
import pandas as pd
from sklearn.preprocessing import LabelEncoder

class PricePredictor:
    """Load model and make predictions."""

    def __init__(self, model_path: str = 'models/best_model.pkl',
                 scaler_path: str = 'models/scaler.pkl'):
        """Initialize predictor with model and scaler."""
        self.model = joblib.load(model_path)
        self.scaler = joblib.load(scaler_path)

        # Initialize encoders (match training)
        self.le_region = LabelEncoder()
        self.le_grape = LabelEncoder()

        # Fit encoders on training data (or save with pickle)
        # For now, using hardcoded list (better: save with model)
        self.regions = ['Alsace', 'Bordeaux', 'Burgundy', 'Loire', 'Rhone', 'Provence']
        self.grapes = ['Cabernet Sauvignon', 'Merlot', 'Pinot Noir', 'Syrah', 'Grenache']

    def predict(self, features: dict) -> dict:
        """
        Make prediction from feature dict.

        Args:
            features: {
                'vintage_year': int,
                'rating': float,
                'region': str,
                'grape_variety': str,
                'food_pairings_count': int
            }

        Returns:
            {
                'predicted_price_eur': float,
                'model': str,
                'input': features
            }
        """
        # Extract & validate
        try:
            vintage = int(features['vintage_year'])
            rating = float(features['rating'])
            region = str(features['region'])
            grape = str(features['grape_variety'])
            pairings = int(features['food_pairings_count'])
        except (KeyError, ValueError, TypeError) as e:
            raise ValueError(f"Invalid input: {e}")

        # Validate ranges
        if not 1900 <= vintage <= 2024:
            raise ValueError(f"Vintage year must be 1900-2024, got {vintage}")
        if not 0 <= rating <= 5:
            raise ValueError(f"Rating must be 0-5, got {rating}")
        if pairings < 0:
            raise ValueError(f"Pairings count must be ≥0, got {pairings}")

        # Create dataframe
        X = pd.DataFrame({
            'vintage_year': [vintage],
            'rating': [rating],
            'region': [region],
            'grape_variety': [grape],
            'food_pairings_count': [pairings]
        })

        # Encode & scale
        X['region'] = self.le_region.fit_transform(X['region'])
        X['grape_variety'] = self.le_grape.fit_transform(X['grape_variety'])

        X_scaled = self.scaler.transform(X)

        # Predict
        prediction = self.model.predict(X_scaled)[0]

        return {
            'predicted_price_eur': round(float(prediction), 2),
            'model': 'XGBoost',
            'input': features
        }


if __name__ == '__main__':
    # Quick test
    predictor = PricePredictor()
    result = predictor.predict({
        'vintage_year': 2018,
        'rating': 4.2,
        'region': 'Bordeaux',
        'grape_variety': 'Cabernet Sauvignon',
        'food_pairings_count': 5
    })
    print(result)
```

### Phase 2: Create serve.py (30 minutes)

```python
# File: src/serve.py
"""Flask web service for wine price prediction."""

from flask import Flask, request, jsonify
from src.predict import PricePredictor

# Initialize Flask app
app = Flask(__name__)

# Load predictor at startup
try:
    predictor = PricePredictor()
    MODEL_LOADED = True
except Exception as e:
    print(f"⚠️  Failed to load model: {e}")
    MODEL_LOADED = False


@app.route('/health', methods=['GET'])
def health():
    """Health check endpoint."""
    return jsonify({
        'status': 'healthy',
        'model_loaded': MODEL_LOADED
    }), 200


@app.route('/predict', methods=['POST'])
def predict():
    """
    Predict wine price.

    Expected JSON:
    {
        "vintage_year": 2018,
        "rating": 4.2,
        "region": "Bordeaux",
        "grape_variety": "Cabernet Sauvignon",
        "food_pairings_count": 5
    }
    """
    if not MODEL_LOADED:
        return jsonify({'error': 'Model not loaded'}), 500

    try:
        # Parse JSON
        data = request.get_json()
        if not data:
            return jsonify({'error': 'No JSON provided'}), 400

        # Predict
        result = predictor.predict(data)
        return jsonify(result), 200

    except ValueError as e:
        return jsonify({'error': str(e)}), 400
    except Exception as e:
        return jsonify({'error': f'Prediction failed: {str(e)}'}), 500


if __name__ == '__main__':
    print("🚀 Starting Wine Price Prediction API on port 9696...")
    app.run(host='0.0.0.0', port=9696, debug=False)
```
