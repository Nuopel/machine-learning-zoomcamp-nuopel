```python
# Cell 3: Price (target) distribution
fig, axes = plt.subplots(1, 2, figsize=(12, 4))

# Histogram
axes[0].hist(df['price'], bins=50, edgecolor='black', alpha=0.7)
axes[0].set_xlabel('Price (EUR)')
axes[0].set_ylabel('Frequency')
axes[0].set_title('Price Distribution')

# Box plot
axes[1].boxplot(df['price'])
axes[1].set_ylabel('Price (EUR)')
axes[1].set_title('Price Box Plot')

plt.tight_layout()
plt.show()

# Statistics
print("Price Statistics:")
print(df['price'].describe())
print(f"Skewness: {df['price'].skew():.2f}")
```

# Cell 8: Food Pairings Count
fig, ax = plt.subplots()

### Phase 5: Correlations (15 minutes)

```python
# Cell 9: Encode categorical for correlation
df_corr = df.copy()
# One-hot encode region and grape_variety (or just drop for correlation)
# Or use label encoding for quick view
from sklearn.preprocessing import LabelEncoder

le_region = LabelEncoder()
le_grape = LabelEncoder()

df_corr['region_encoded'] = le_region.fit_transform(df_corr['region'])
df_corr['grape_encoded'] = le_grape.fit_transform(df_corr['grape_variety'])

# Correlation with price
corr_with_price = df_corr[['vintage_year', 'rating', 'region_encoded',
                             'grape_encoded', 'food_pairings_count', 'price']].corr()['price'].sort_values(ascending=False)

print("Correlation with Price:")
print(corr_with_price)

# Heatmap
fig, ax = plt.subplots(figsize=(8, 6))
sns.heatmap(df_corr[['vintage_year', 'rating', 'region_encoded',
                      'grape_encoded', 'food_pairings_count', 'price']].corr(),
            annot=True, fmt='.2f', cmap='coolwarm', ax=ax)
ax.set_title('Feature Correlation Heatmap')
plt.tight_layout()
plt.show()
```
